UnityEngine.Application.runInBackground = true;
var sb = new System.Text.StringBuilder();

System.Type handT = null, popT = null;
foreach (var asm in System.AppDomain.CurrentDomain.GetAssemblies())
{
    if (handT == null) handT = asm.GetType("_DreamKeep.Presentation.UI_HandView");
    if (popT == null) popT = asm.GetType("_DreamKeep.Presentation.UI_PopTextRoot");
}
if (handT == null || popT == null) return "TYPES_MISSING";
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Public;
var hands = UnityEngine.Object.FindObjectsOfType(handT, true);
sb.AppendLine("handCount=" + hands.Length);
if (hands.Length == 0) return sb.ToString() + " NO_HANDVIEW";
var hand = hands[0] as UnityEngine.MonoBehaviour;
var pops = UnityEngine.Object.FindObjectsOfType(popT, true);
if (pops.Length == 0) return sb.ToString() + " NO_POPTEXTROOT";
var pop = pops[0] as UnityEngine.MonoBehaviour;

var fHover = handT.GetField("_hoveredCard", flags);
var fDrag = handT.GetField("_draggedCard", flags);
var fTip = handT.GetField("_tipCard", flags);
var fPop = handT.GetField("_popTextRoot", flags);
var fHandList = handT.GetField("_hand", flags);
var mTick = handT.GetMethod("TickAffixTips", flags);
var fShown = popT.GetField("_shownTips", flags);
var fReq = popT.GetField("_tipsRequester", flags);
sb.AppendLine("popInjected=" + (fPop.GetValue(hand) != null) + " shownField=" + (fShown != null) + " reqField=" + (fReq != null));

var handList = fHandList.GetValue(hand) as System.Collections.IList;
sb.AppendLine("handList=" + (handList == null ? -1 : handList.Count));
if (handList == null || handList.Count == 0) return sb.ToString() + " HAND_EMPTY";

UnityEngine.Component cardWith = null, cardWithout = null;
var counts = new System.Text.StringBuilder();
for (int i = 0; i < handList.Count; i++)
{
    var card = (UnityEngine.Component)handList[i];
    var model = card.GetType().GetProperty("Card", flags).GetValue(card, null);
    var lines = model.GetType().GetMethod("GetDescriptionLines", flags).Invoke(model, null) as System.Collections.IEnumerable;
    int n = 0;
    foreach (var line in lines)
        if (line.GetType().GetField("Item2").GetValue(line) != null) n++;
    counts.Append(n + " ");
    if (n > 0 && cardWith == null) cardWith = card;
    if (n == 0 && cardWithout == null) cardWithout = card;
}
sb.AppendLine("affixCounts=" + counts);
if (cardWith == null) return sb.ToString() + " NO_AFFIX_CARD";

hand.enabled = false; // 冻结 Update：各步裁决一律手动调 TickAffixTips，断言之间互不干扰

System.Func<int> activeCount = delegate()
{
    int a = 0;
    var list = fShown.GetValue(pop) as System.Collections.IList;
    for (int i = 0; i < list.Count; i++)
        if (((UnityEngine.Component)list[i]).gameObject.activeSelf) a++;
    return a;
};
System.Func<string> snapshot = delegate()
{
    var list = fShown.GetValue(pop) as System.Collections.IList;
    return "shown=" + list.Count + " active=" + activeCount()
        + " tipCardIsCard=" + (fTip.GetValue(hand) == cardWith)
        + " requesterIsCard=" + (fReq.GetValue(pop) == (object)cardWith);
};

// A) 悬停：显示，整列挂在牌名下，第一条落在 AffixTipAnchor 上
fDrag.SetValue(hand, null);
fHover.SetValue(hand, cardWith);
mTick.Invoke(hand, null);
sb.AppendLine("A_hover: " + snapshot());
var listA = fShown.GetValue(pop) as System.Collections.IList;

if (listA.Count > 0)
{
    var anchor = (UnityEngine.Vector3)cardWith.GetType().GetProperty("AffixTipAnchor", flags).GetValue(cardWith, null);
    var firstA = (UnityEngine.Component)listA[0];
    sb.AppendLine("A_anchorErr=" + UnityEngine.Vector3.Distance(firstA.transform.position, anchor).ToString("F4"));
    for (int i = 0; i < listA.Count; i++)
        if (((UnityEngine.Component)listA[i]).transform.parent != cardWith.transform) parentOk = false;
}
sb.AppendLine("A_parentAllCard=" + parentOk);

// B) 抓起拖拽（本次改动）：悬停已撤、只剩拖拽身份，气泡必须仍在（旧判据此处会收掉）
fHover.SetValue(hand, null);
fDrag.SetValue(hand, cardWith);
mTick.Invoke(hand, null);
sb.AppendLine("B_drag: " + snapshot() + " (expect shown>0 active>0 tipCardIsCard=True)");

// C) 拖拽期间随卡移动：整列是卡的孩子，卡挪多少气泡就挪多少
if (listA.Count > 0)
{
    var firstC = (UnityEngine.Component)listA[0];
    var p0 = firstC.transform.position;
    var old = cardWith.transform.position;
    cardWith.transform.position = old + new UnityEngine.Vector3(50f, 30f, 0f);
    sb.AppendLine("C_followDelta=" + (firstC.transform.position - p0).ToString("F2") + " (expect 50.00, 30.00, 0.00)");
    cardWith.transform.position = old;
}

// D) 松手受理打出：拖拽身份清掉、牌已交给出牌区（不再悬停），气泡当帧收起
fDrag.SetValue(hand, null);
fHover.SetValue(hand, null);
mTick.Invoke(hand, null);
sb.AppendLine("D_afterPlay: " + snapshot() + " tipCardNull=" + (fTip.GetValue(hand) == null) + " (expect shown=0 active=0 tipCardNull=True)");

// E) 松手驳回退回后按悬停事实：指针仍压在这张牌上 → 再次显示
fHover.SetValue(hand, cardWith);
mTick.Invoke(hand, null);
sb.AppendLine("E_afterReject: " + snapshot());

// F) 无关键词的牌：什么都不挂
if (cardWithout != null)
{
    fHover.SetValue(hand, cardWithout);
    mTick.Invoke(hand, null);
    sb.AppendLine("F_noAffix: shown=" + (fShown.GetValue(pop) as System.Collections.IList).Count
        + " active=" + activeCount() + " (expect 0/0)");
}

// G) 复原现场：清交互身份、收起气泡、恢复手牌区 Update
fHover.SetValue(hand, null);
fDrag.SetValue(hand, null);
mTick.Invoke(hand, null);
hand.enabled = true;
sb.AppendLine("G_restore: " + snapshot() + " handEnabled=" + hand.enabled);
return sb.ToString();
